  <script>
    // สลับหน้าเว็บ
    function switchPage(pageId, element) {
      const pages = document.querySelectorAll('.page-section');
      pages.forEach(page => page.classList.remove('active'));

      const navButtons = document.querySelectorAll('.nav-btn');
      navButtons.forEach(btn => btn.classList.remove('active'));

      document.getElementById(pageId).classList.add('active');
      if (element) element.classList.add('active');
    }

    // คำนวณตารางความจริง
    function generateTruthTable() {
      const op = document.getElementById("operator").value;
      const inputs = [
        { p: true, q: true },
        { p: true, q: false },
        { p: false, q: true },
        { p: false, q: false }
      ];

      let html = `<table><tr><th>P</th><th>Q</th><th>P ${op} Q</th></tr>`;

      inputs.forEach(item => {
        let res = false;
        if (op === "AND") res = item.p && item.q;
        else if (op === "OR") res = item.p || item.q;
        else if (op === "XOR") res = item.p !== item.q;
        else if (op === "IMPLIES") res = !item.p || item.q;

        const pStyle = item.p ? 'color:#00d4ff;' : 'color:#ff4d4d;';
        const qStyle = item.q ? 'color:#00d4ff;' : 'color:#ff4d4d;';
        const resStyle = res ? 'color:#00d4ff; font-weight:bold;' : 'color:#ff4d4d; font-weight:bold;';

        html += `<tr>
          <td style="${pStyle}">${item.p}</td>
          <td style="${qStyle}">${item.q}</td>
          <td style="${resStyle}">${res}</td>
        </tr>`;
      });

      html += `</table>`;
      document.getElementById("tableContainer").innerHTML = html;
    }

    // ประมวลผล AI Logic
    function evaluateAI() {
      const income = document.getElementById("income").value === "true";
      const credit = document.getElementById("creditHistory").value === "true";
      const resultBox = document.getElementById("aiResult");

      let decision = "";
      if (income && credit) {
        decision = "ผลการอนุมัติ: <span style='color:#00d4ff;'>ผ่านการอนุมัติ (Approved)</span>";
      } else if (income && !credit) {
        decision = "ผลการอนุมัติ: <span style='color:#ff66cc;'>ต้องมีผู้ค้ำประกัน (Conditional)</span>";
      } else {
        decision = "ผลการอนุมัติ: <span style='color:#ff4d4d;'>ไม่ผ่านการอนุมัติ (Rejected)</span>";
      }

      resultBox.style.display = "block";
      resultBox.innerHTML = decision;
    }

    // โหลดตารางเริ่มต้น
    generateTruthTable();
  </script>
