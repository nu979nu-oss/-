/* =====================================================
   PAGE NAVIGATION
===================================================== */

function showPage(pageName) {

    const pages =
        document.querySelectorAll(".page");

    pages.forEach(page => {
        page.classList.remove("active");
    });

    document
        .getElementById(pageName)
        .classList.add("active");

    window.scrollTo({
        top: 0,
        behavior: "smooth"
    });

    if (pageName === "cart") {
        displayCart();
    }

}


/* =====================================================
   AI RECOMMENDATION
===================================================== */

function recommend() {

    const taste =
        document.getElementById("taste").value;

    let result = "";

    if (taste === "sweet") {

        result =
            "🍰 AI แนะนำ: Strawberry Cake " +
            "เพราะมีรสหวานและครีมเข้มข้น";

    }

    else if (taste === "chocolate") {

        result =
            "🍫 AI แนะนำ: Chocolate Brownie " +

/* =====================================================
   DISPLAY CART
===================================================== */

function displayCart() {

    const container =
        document.getElementById("cartItems");

    if (cart.length === 0) {

        container.innerHTML = `

            <div class="empty">

                <h2>
                    ตะกร้ายังว่าง 🍪
                </h2>

                <br>

                <p>
                    ลองเลือกขนมที่คุณชื่นชอบก่อน
                </p>

            </div>

        `;

        document
            .getElementById("total")
            .innerText = "฿0";

        analyzeCart();

        return;
    }


    let html = "";
    let total = 0;


    cart.forEach((item, index) => {

        total += item.price;

        html += `

            <div class="item">

                <span class="item-name">

                    ${index + 1}.
                    ${item.name}

                </span>

                <span class="item-price">

                    ฿${item.price}

                </span>

                <button
                    class="delete"
                    onclick="removeItem(${index})">

                    ลบ

                </button>

            </div>

        `;

    });


    container.innerHTML = html;


    document
        .getElementById("total")
        .innerText = "฿" + total;


    analyzeCart();

}


/* =====================================================
   REMOVE ITEM
===================================================== */

function removeItem(index) {

    cart.splice(index, 1);

    saveCart();

    updateCartCount();

    displayCart();

}


/* =====================================================
   CHECKOUT
===================================================== */

function checkout() {

    if (cart.length === 0) {

        alert(
            "⚠️ กรุณาเลือกขนมก่อนสั่งซื้อ"
        );

        return;
    }


    let total = cart.reduce(
        (sum, item) =>
            sum + item.price,
        0
    );


    alert(

        "🎉 สั่งซื้อสำเร็จ!\n\n" +

        "จำนวนสินค้า: " +
        cart.length +
        " รายการ\n" +

        "ยอดรวม: ฿" +
        total +
        "\n\n" +

        "ขอบคุณที่ใช้บริการ Sweet AI 🍰"

    );


    cart = [];

    localStorage.removeItem(
        "sweetAIcart"
    );

    updateCartCount();

    displayCart();

}

function analyzeCart() {

    const result =
        document.getElementById("aiAnalysis");

/* =====================================================
   FACTORIAL
===================================================== */

function factorial(num) {

    let result = 1;

    for (
        let i = 2;
        i <= num;
        i++
    ) {

        result *= i;

    }

    return result;

}


/* =====================================================
   COMBINATION
===================================================== */

function calculateCombination() {

    const n =
        parseInt(
            document.getElementById("n").value
        );

    const r =
        parseInt(
            document.getElementById("r").value
        );


    const resultBox =
        document.getElementById(
            "combinationResult"
        );


    if (
        isNaN(n) ||
        isNaN(r) ||
        r > n ||
        r < 0 ||
        n < 0
    ) {

        resultBox.innerHTML =

            "⚠️ กรุณากรอกค่าให้ถูกต้อง " +
            "โดย r ต้องไม่มากกว่า n";

        return;

    }


    const result =

        factorial(n) /
        (
            factorial(r) *
            factorial(n - r)
        );


    resultBox.innerHTML =

        `🎯 C(${n}, ${r}) = ${result} ชุด`;

}
